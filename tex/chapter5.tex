\chapter{نتیجه‌گیری و پیشنهادها}
\label{ch:conclusion}

\section{مرور دستاوردهای پژوهش}

هدف اصلی این رساله، توسعه‌ی یک بستر شبیه‌سازی بلادرنگ و معتبر برای موتورهای صفحه‌ای با شناوری مغناطیسی (\lr{MLPM}) بود؛ بستری که بتواند به‌عنوان یک سکوی آزمون مجازی، امکان طراحی، پیاده‌سازی و ارزیابی الگوریتم‌های کنترلی را بدون نیاز به سخت‌افزار فیزیکی فراهم سازد. این هدف در قالب یک چارچوب یکپارچه متشکل از مدل‌سازی دینامیکی، شبیه‌سازی فیزیکی، کنترل تطبیقی و تحلیل سینماتیکی دنبال شد و دستاوردهای حاصل در ادامه خلاصه می‌گردند.

در گام نخست، \textbf{مدل ریاضی سیستم} بر اساس ساختار معرفی‌شده در پژوهش \cite{RN10} استخراج گردید. معادلات حاکم بر نیرو و گشتاور در قالب ماتریس سینماتیک مستقیم $\Gamma$ تدوین شد و با بهره‌گیری از وارون شبه‌مور-پنروز $\Gamma^\dagger$، سازوکار تبدیل پیچش مطلوب به جریان‌های مرجع سیم‌پیچ‌ها و بالعکس، با در نظر گرفتن قیود عملی، پیاده‌سازی گردید.

در گام دوم، \textbf{محیط شبیه‌سازی فیزیکی} با تلفیق موتور فیزیک \lr{Gazebo Harmonic} و چارچوب رباتیک \lr{ROS 2 Jazzy} ایجاد شد. در این بستر، مدل دقیق بخش متحرک (شامل جرم، اینرسی و هندسه) تعریف گردید و سازوکارهای ارتباطی برای تبادل پیچش کنترلی و بازخورد وضعیت از طریق تاپیک‌های \lr{ROS 2} پیاده‌سازی شد. هم‌چنین، یک گره همگام‌ساز با منطق نگهدارنده‌ی مرتبه‌ی صفر (\lr{Zero-Order Hold}) برای تضمین پایداری سیگنال‌های گسسته در تعامل با شبیه‌ساز پیوسته توسعه یافت.

در گام سوم، \textbf{کنترل‌کننده‌ی فازی-\lr{PID}} با ۲۵ قاعده و توابع عضویت گوسی طراحی گردید که بهره‌های تناسبی، انتگرالی و مشتق را به‌طور بلادرنگ بر اساس خطا و مشتق خطا تنظیم می‌کند. یک ماشین حالت دوفازی نیز برای مدیریت ایمن گذار از وضعیت «مستقر روی زمین» به «معلق» پیاده‌سازی شد که از بروز ناپایداری‌های ناشی از تماس اولیه جلوگیری می‌کند. علاوه بر این، یک ماژول برنامه‌ریز مسیر مبتنی بر چندجمله‌ای‌های مرتبه پنجم برای تولید پروفایل‌های هموار موقعیت، سرعت و شتاب به کار گرفته شد.

در گام چهارم، \textbf{تحلیل تکینگی سینماتیکی} با استفاده از تجزیه مقادیر تکین ماتریس $\Gamma$ و جستجوی شبکه‌ای انجام شد تا نواحی ایمن و بحرانی فضای کاری شناسایی گردند.

\section{یافته‌های کلیدی}

نتایج حاصل از اعتبارسنجی تجربی چارچوب پیشنهادی، دستاوردهای زیر را به‌طور کمّی مستند می‌سازند:

\begin{enumerate}
	\item \textbf{عملکرد ردیابی در آزمون‌های پله:} کنترل‌کننده‌ی فازی-\lr{PID} در محورهای افقی $X$ و $Y$ به زمان خیز حدود $0.37$ ثانیه، زمان نشست نزدیک به $1.12$ ثانیه و فراجهش کمتر از $1.25\%$ دست یافت. در محور عمودی $Z$، با وجود چالش گرانش، زمان نشست $12.8$ ثانیه با خطای ماندگار در مرتبه‌ی $10^{-6}$ متر حاصل شد. درجات آزادی زاویه‌ای $\alpha$ و $\beta$ نیز با فراجهش $11$ تا $16\%$ و خطای ماندگار در مرتبه‌ی $10^{-8}$ رادیان پایدار شدند.
	
	\item \textbf{مزیت مکانیزم تطبیقی فازی:} مقایسه با یک کنترلر \lr{PID} کلاسیک با ضرایب ثابت نشان داد که کنترلر فازی-\lr{PID} فراجهش را $43\%$، خطای حالت ماندگار را $59.7\%$ و شاخص‌های خطای انتگرالی نظیر IAE و ITAE را به‌ترتیب $26\%$ و $34.1\%$ کاهش می‌دهد. این بهبودها مستقیماً به توانایی سیستم فازی در تنظیم لحظه‌ای بهره‌ها نسبت داده شد.
	
	\item \textbf{تخصیص جریان تحت قیود واقعی:} با اعمال هم‌زمان اشباع جریان ($\pm 9$ آمپر) و محدودیت نرخ تغییرات ($80$ آمپر بر ثانیه)، زنجیره‌ی تخصیص کنترلی جریان‌های پایدار، متقارن و عاری از تکینگی تولید کرد. در تمامی آزمون‌ها، جریان‌ها در محدوده‌ی مجاز باقی ماندند و هیچ‌گاه به کران‌های اشباع نرسیدند.
	
	\item \textbf{سلامت سینماتیکی فضای کاری:} تحلیل عدد شرط نشان داد که با تنظیم $\gamma$ در حدود $\pi/4$، عدد شرط ماتریس $\Gamma$ به کمینه‌ی $18$ می‌رسد و در کل صفحه‌ی افقی از $34$ فراتر نمی‌رود. زوایای رول و پیچ تا $0.06$ رادیان نیز عدد شرط را حداکثر به $20$ می‌رسانند. خارج از این ناحیه، به‌ویژه در نزدیکی $\gamma = \pm\pi/2$ و $\pm\pi$، عدد شرط به شدت افزایش یافته و شبیه‌سازی ناپایدار می‌شود.
	
	\item \textbf{ردیابی مسیر منهتن:} سیستم قادر به تعقیب یک مسیر پلکانی چندبخشی در صفحه بود. با کاهش زمان هر قطعه به $0.5$ ثانیه، انباشتگی خطا مشاهده شد، اما در $1.5$ ثانیه به ازای هر قطعه، مسیر تقریباً بی‌نقص دنبال گردید. این نتایج، محدودیت سرعت ذاتی سیستم تحت قیود موجود را مشخص ساختند.
\end{enumerate}

\section{محدودیت‌ها}

علیرغم دستاوردهای فوق، پژوهش حاضر با محدودیت‌هایی همراه است که می‌بایست در تفسیر نتایج و توسعه‌های آتی مد نظر قرار گیرند:

\begin{itemize}
	\item \textbf{ماهیت شبیه‌سازی:} کلیه‌ی نتایج در یک محیط شبیه‌سازی فیزیکی (\lr{Gazebo}) به دست آمده‌اند. اگرچه تلاش شده است با اعمال قیود سخت‌افزاری، رفتار سیستم به واقعیت نزدیک شود، اما صحت‌سنجی نهایی نیازمند پیاده‌سازی بر روی یک نمونه‌ی آزمایشگاهی واقعی است.
	
	\item \textbf{مدل ساده‌شده‌ی عملگرها:} روابط نیرو و گشتاور مورد استفاده، یک مدل تحلیلی شناسایی‌شده در ارتفاع ثابت هستند و در سایر ارتفاعات ممکن است از دقت کمتری برخوردار باشند.
	
	\item \textbf{تحلیل ناکامل جریان‌ها:} بررسی جریان‌های سیم‌پیچ‌ها در این پژوهش به یک بازبینی کیفی محدود شد. برای استخراج قواعد حاکم بر توزیع فضایی جریان‌ها، نیاز به مجموعه‌داده‌های گسترده‌تری از مانورهای ترکیبی است.
	
	\item \textbf{محدودیت زاویه‌ای $\gamma$:} عملکرد سیستم صرفاً در بازه‌ی امن $\gamma \approx \pi/4$ ارزیابی شد و برای زوایای بزرگ‌تر، به‌ویژه در نزدیکی تکینگی‌ها، راهکاری ارائه نگردید.
\end{itemize}

\section{پیشنهادها برای ادامه‌ی کار}

بر اساس یافته‌ها و محدودیت‌های ذکر شده، مسیرهای زیر برای پژوهش‌های آتی پیشنهاد می‌گردد:

\begin{enumerate}
	\item \textbf{انتقال به سخت‌افزار واقعی:} نخستین و مهم‌ترین گام، پیاده‌سازی چارچوب کنترلی حاضر بر روی یک نمونه‌ی فیزیکی \lr{MLPM} و مقایسه‌ی نتایج تجربی با داده‌های شبیه‌سازی است. این امر اعتبار نهایی بستر توسعه‌یافته را اثبات خواهد کرد.
	
	\item \textbf{بهبود مدل‌سازی:} جایگزینی مدل تحلیلی حاضر با یک مدل مبتنی بر داده‌های تجربی یا استفاده از روش‌های شناسایی سیستم می‌تواند دقت نگاشت $\Gamma$ را به‌ویژه در ارتفاعات مختلف افزایش دهد.
	
	\item \textbf{توسعه‌ی کنترلرهای پیشرفته:} بستر شبیه‌سازی ایجادشده، امکان آزمودن کنترلرهای مدرن‌تری مانند کنترل پیش‌بین (\lr{MPC})، کنترل تطبیقی و روش‌های مبتنی بر یادگیری تقویتی را فراهم می‌سازد.
	
	\item \textbf{راهبردهای عبور از تکینگی:} طراحی مسیرهای حرکتی که بتوانند با حفظ پایداری از نواحی تکین عبور کنند، یک گام ضروری برای بهره‌برداری کامل از فضای کاری سیستم است.
	
	\item \textbf{بهینه‌سازی تخصیص جریان:} تحلیل جامع الگوهای جریان و بهینه‌سازی توزیع آن‌ها برای کاهش مصرف انرژی و افزایش یکنواختی نیرو می‌تواند کارایی سیستم را بهبود بخشد.
	
	\item \textbf{توسعه‌ی واسط کاربری:} ایجاد یک واسط گرافیکی برای کنترل بی‌درنگ مسیر و پایش وضعیت سیستم، بستر را برای کاربران غیرمتخصص نیز قابل استفاده خواهد کرد.
\end{enumerate}

\section{سخن پایانی}

این پژوهش نشان داد که تلفیق یک موتور فیزیک دقیق، یک چارچوب رباتیکی منعطف و یک کنترل‌کننده‌ی تطبیقی، می‌تواند یک سکوی آزمون مجازی قدرتمند برای سیستم‌های \lr{MLPM} فراهم آورد. بستر توسعه‌یافته نه تنها امکان ارزیابی الگوریتم‌های کنترلی را بدون نیاز به سخت‌افزار مهیا می‌کند، بلکه با شبیه‌سازی وفادار محدودیت‌های عملی، فاصله‌ی میان طراحی نظری و پیاده‌سازی فیزیکی را کاهش می‌دهد. امید می‌رود که این بستر، زمینه‌ساز پژوهش‌های گسترده‌تر در حوزه‌ی کنترل حرکت‌های دقیق و بدون تماس باشد.
